---
title: recur
description: A bash script that runs a passed command in the root of all directories it finds within a passed depth
---

`recur` runs a passed command in all directories found within a depth from a root directory.

```bash title="recur"
#!/usr/bin/env bash

RED='\033[0;31m'
GREEN='\033[0;32m'
RESET='\033[0m'

OPTS=`getopt --options hr:d:c:q --longoptions help,relative-to:,max-depth:,command:,quiet -n 'parse-options' -- "$@"`

help() {
  echo -e "${GREEN}recur --relative-to --max-depth --command${RESET}"
  echo -e "-h | --help        : Help"
  echo -e "-r | --relative-to : The root directory that the search will begin in, default ."
  echo -e "-d | --max-depth   : The max depth that the program will search for git directories, default 1"
  echo -e "-c | --command     : The command to run at the root of each directory (i.e. \"git pull --prune\", \"git log | head -1\")"
  echo -e "-q | --quiet       : Suppresses printing the found directory names"
}

if [ $? != 0 ] ; then echo "Failed parsing options." >&2 ; exit 1 ; fi

eval set -- "$OPTS"

RELATIVE_TO="."
MAX_DEPTH=1
COMMAND=""
QUIET=

while true; do
  case "$1" in
    -h | --help ) help; exit ;;
    -s | --relative-to ) RELATIVE_TO=$2; shift; shift ;;
    -e | --max-depth ) MAX_DEPTH=$2; shift; shift ;;
    -c | --command ) COMMAND=$2; shift; shift ;;
    -q | --quiet) QUIET='1'; shift ;;
    -- ) shift; break ;;
    * ) break ;;
  esac
done

if [ -z "$COMMAND" ]; then
	echo -e "${RED}No command specified with --command${RESET}"
	echo -e "${GREEN}Listing directories...${RESET}"
fi

dirs=($(find $RELATIVE_TO -mindepth 1 -maxdepth $MAX_DEPTH -type d))

if [ -z "$QUIET" ]; then
	echo -n "Getting directories in $(realpath $RELATIVE_TO)"
	echo " (found ${#dirs[@]})"
fi

for dir in "${dirs[@]}"; do
  if [ -d "${dir}" ]; then
    cd "${dir}"
    
      if [ -z "$QUIET" ]; then
        echo -e "${GREEN}${dir}${RESET}"
      fi

      if [ -n "$COMMAND" ]; then
        eval $COMMAND
      fi
    cd -> /dev/null
  fi
done
```
